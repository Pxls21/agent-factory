"""K170 premise: how the real lock, SBOM and provenance files look to the line parsers and to YAML.

Run from the repository root: python3 <this file>. Reads only.
"""
import collections
import hashlib
import json
import re
import sys
from pathlib import Path

import yaml

sys.path.insert(0, "scripts")
import vendored_manifest as vm  # noqa: E402


def digest(mapping) -> str:
    return hashlib.sha256(json.dumps(sorted(mapping.items())).encode()).hexdigest()[:16]


# 1. What the lock's line parser does with each line.
cats = collections.Counter()
skipped = collections.Counter()
section = component = ""
for line in vm.LOCK_PATH.read_text(encoding="utf-8").splitlines():
    if not line.strip() or line.lstrip().startswith("#"):
        cats["blank/comment"] += 1
    elif not line.startswith(" ") and line.endswith(":"):
        cats["section"] += 1
        section, component = line[:-1], ""
    elif re.fullmatch(r"  ([a-z0-9][a-z0-9-]*):", line):
        cats["component"] += 1
        component = line.strip()[:-1]
    elif re.fullmatch(r"    (repository|commit|binary_sha256|asset_sha256):\s*(.+)", line) and component:
        cats["pin-field"] += 1
    elif line.startswith("    ") and ":" not in line:
        cats["refused"] += 1
    else:
        cats["skipped"] += 1
        skipped[section or "(top level)"] += 1
print("lock lines by parse_lock branch:", dict(sorted(cats.items())))
print("skipped lines per section:", dict(sorted(skipped.items())))

# 2. The parsers' results against YAML's reading, with the same extraction rule.
lock = vm.parse_lock(vm.LOCK_PATH)
sbom = vm.parse_sbom(vm.SBOM_PATH)
ydoc = yaml.safe_load(vm.LOCK_PATH.read_text(encoding="utf-8"))
ylock, norepo, nopin, twopins, repos = {}, [], [], [], collections.Counter()
for sec, body in ydoc.items():
    if not isinstance(body, dict):
        print(f"lock top-level non-mapping: {sec}={type(body).__name__}")
        continue
    for comp, vals in body.items():
        pins = [k for k in ("commit", "binary_sha256", "asset_sha256") if k in vals]
        if len(pins) > 1:
            twopins.append(f"{sec}.{comp}:{'+'.join(pins)}")
        if "repository" not in vals:
            norepo.append(f"{sec}.{comp}")
        elif not pins:
            nopin.append(f"{sec}.{comp}")
        else:
            key = vm.normalize_repo(str(vals["repository"]))
            repos[key] += 1
            ylock[key] = (str(vals[pins[0]]), f"{sec}.{comp}")
print(f"parse_lock: {len(lock)} entries, digest {digest(lock)}; yaml extraction equal: {ylock == lock}")
print("lock components without repository:", norepo)
print("lock components with a repository and no pin key:", nopin)
print("lock components with two pin keys:", twopins)
print("lock repositories pinned twice:", [r for r, n in repos.items() if n > 1])
print("lock pin/repository value types:", dict(collections.Counter(
    type(v[k]).__name__ for b in ydoc.values() if isinstance(b, dict) for v in b.values()
    for k in ("repository", "commit", "binary_sha256", "asset_sha256") if k in v)))
sdoc = yaml.safe_load(vm.SBOM_PATH.read_text(encoding="utf-8"))
ysbom = {vm.normalize_repo(str(c["repository"])): (str(c.get("commit") or c.get("digest")), c["name"])
         for c in sdoc["components"]}
print(f"parse_sbom: {len(sbom)} entries, digest {digest(sbom)}; yaml extraction equal: {ysbom == sbom}; "
      f"components {len(sdoc['components'])}; keys {sorted({k for c in sdoc['components'] for k in c})}")


# 3. Duplicate mapping keys at any depth.
class Unique(yaml.SafeLoader):
    pass


dups = []


def mapping(loader, node, deep=False):
    loader.flatten_mapping(node)
    seen = set()
    for key_node, _ in node.value:
        key = loader.construct_object(key_node, deep=deep)
        if key in seen:
            dups.append((key, key_node.start_mark.line + 1))
        seen.add(key)
    return yaml.SafeLoader.construct_mapping(loader, node, deep)


Unique.add_constructor(yaml.resolver.BaseResolver.DEFAULT_MAPPING_TAG, mapping)
for path in (vm.LOCK_PATH, vm.SBOM_PATH):
    dups.clear()
    yaml.load(path.read_text(encoding="utf-8"), Loader=Unique)
    print(f"{path}: duplicate keys at any depth: {dups}")

# 4. The provenance table: where the walk ends today.
lines = vm.PROVENANCE_PATH.read_text(encoding="utf-8").splitlines()
header = next(i for i, l in enumerate(lines) if vm.markdown_cells(l)[:2] == ["Source (pxls21/sandbox-kit)", "Here"])
end = header + 2
while end < len(lines) and lines[end].startswith("|"):
    end += 1
print(f"provenance: header line {header + 1}, {end - header - 2} rows, the walk stops at line {end + 1} "
      f"({lines[end]!r}); roots {len(vm.parse_provenance(vm.PROVENANCE_PATH)[1])}")
